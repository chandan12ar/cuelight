# M1 Advice Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn Cuelight from "what happened" into "what to change": fix-it cards with a cause for every failure, a one-line run verdict, "usual" time/cost per kind of agent learned from past runs, shown in a new Advice tab and printed by `/cuelight:open advise` and `/cuelight:open estimate`.

**Architecture:** Two new pure-Python modules on top of what the builder already computes. `orchestra/advice.py` is a pure function of a `Run` (plus prices and the usual numbers already attached to each agent) that returns a verdict and ordered cards. `orchestra/baselines.py` reads past sessions with the existing `RunBuilder`, keeps numbers only in a cache in the state dir, and answers "what does this (agent type, model) usually take". Built-in list prices (`orchestra/listprices.py`) make dollars show by default. The UI is one new script, `orchestra/static/advice.js`.

**Tech Stack:** Python 3.9+ standard library only; vanilla JS (no build step); `unittest`; Node for the UI tests (skipped when absent).

**Spec:** `docs/superpowers/specs/2026-10-12-m1-advice-design.md` (read it with this plan).

```python
"""Anthropic's public list prices: the default when you have no price file of your own.

Per million tokens, checked on AS_OF against SOURCE. Every figure priced from this table is
labelled "list price, as of AS_OF"; your own price file always wins, and ORCHESTRA_PRICES=off
turns cost off altogether.

Two choices keep it from understating, as orchestra.pricing promises:
* cache writes are priced at the one-hour rate (2x input). Claude Code writes most of its cache
  for an hour (72% of cache writes in the transcripts this was checked against); a five-minute
  write (1.25x input) is overstated.
* Claude Haiku 5.5 is priced at its rate for prompts over 100K tokens; shorter prompts cost a fifth.
Models not listed (older Opus 4 and Sonnet 4.5 and before) stay unpriced and are reported as
such, rather than guessed.
"""

from typing import Dict, Tuple

from orchestra.pricing import PriceTable

AS_OF = "2026-10-12"
SOURCE = "https://platform.claude.com/docs/en/about-claude/pricing"

# The cheaper model orchestra.advice suggests for an agent that only read and searched, by family:
# (the alias for an agent's definition, the model id it is priced as).
CHEAPER: Dict[str, Tuple[str, str]] = {"opus": ("sonnet", "claude-sonnet-5-5"),
                                       "sonnet": ("haiku", "claude-haiku-5-5")}


def _p(input_: float, output: float, cache_read: float) -> Dict[str, float]:
    return {"input": input_, "output": output, "cache_read": cache_read, "cache_create": input_ * 2}


# First match wins, so a model's own row comes before its family's.
MODELS = (
    ("claude-fable-5-1*", _p(10.0, 50.0, 0.25)),
    ("claude-mythos-5-1*", _p(10.0, 50.0, 0.25)),
    ("claude-fable-5*", _p(10.0, 50.0, 1.0)),
    ("claude-mythos-5*", _p(10.0, 50.0, 1.0)),
    ("claude-opus-5-5*", _p(4.0, 20.0, 0.20)),
    ("claude-opus-5*", _p(5.0, 25.0, 0.50)),
    ("claude-opus-4-[678]*", _p(5.0, 25.0, 0.50)),
    ("claude-sonnet-5-5*", _p(2.0, 10.0, 0.20)),
    ("claude-sonnet-5*", _p(2.0, 10.0, 0.20)),
    ("claude-sonnet-4-6*", _p(3.0, 15.0, 0.30)),
    ("claude-haiku-5-5*", _p(0.50, 2.50, 0.05)),
    ("claude-haiku-4-5*", _p(1.0, 5.0, 0.10)),
)


def table() -> PriceTable:
    t = PriceTable.parse({"currency": "USD", "models": dict(MODELS)})
    t.source, t.as_of = "list", AS_OF
    return t
```

```python
"""Fix-it cards and the run verdict (orchestra.advice), on hand-built runs."""

import unittest
from collections import OrderedDict

from orchestra import advice
from orchestra import constants as C
from orchestra.errors import ErrorLog
from orchestra.model import Agent, Round, Run, ToolCall
from orchestra.waste import WasteLog


def agent(aid="a1", status=C.COMPLETED, start=0.0, end=600.0, desc=None, **kw):
    kw.setdefault("model", "claude-sonnet-5")
    kw.setdefault("agent_type", "general-purpose")
    return Agent(agent_id=aid, description=desc or "Agent " + aid, status=status,
                 rounds=[Round(started_at=start, ended_at=end, status=status)], **kw)


def calls(*specs):
    return [ToolCall(name=n, target=t, timestamp=at, ok=ok) for n, t, at, ok in specs]


def errlog(api=(), calls=(), timeouts=()):
    log = ErrorLog()
    log.api = [dict(e) for e in api]
    log.calls = [list(c) for c in calls]
    log.timeouts = [dict(t) for t in timeouts]
    return log


def api_error(at, kind="rate_limit", text="Usage limit reached"):
    return {"at": at, "kind": kind, "status": 429, "text": text, "resumed_at": None, "repeats": 1}


def run(agents, **kw):
    return Run(session_id="s1", agents=list(agents), **kw)


def cards(r, now=10_000.0, table=None):
    return advice.compute(r, now, table)["cards"]


def rule(r, name, **kw):
    return [c for c in cards(r, **kw) if c["rule"] == name]


class TestFailureCauses(unittest.TestCase):
    def fail(self, **kw):
        out = cards(run([agent(status=C.FAILED, **kw)]))
        self.assertEqual(len(out), 1)
        self.assertEqual((out[0]["rule"], out[0]["severity"]), ("failure", "failure"))
        return out[0]

    def test_an_api_error_at_the_end_names_the_kind_and_its_fix(self):
        c = self.fail(errors=errlog(api=[api_error(590.0)]))
        self.assertIn("usage limit", c["title"].lower())
        self.assertIn("reset", c["fix"]["text"])

    def test_an_old_api_error_is_not_the_cause(self):
        c = self.fail(errors=errlog(api=[api_error(10.0)]))
        self.assertNotIn("usage limit", c["title"].lower())

    def test_an_expired_login(self):
        c = self.fail(errors=errlog(api=[api_error(595.0, kind="authentication_failed", text="401")]))
        self.assertIn("/login", c["fix"]["text"])

    def test_stuck_on_the_same_failing_call(self):
        log = errlog(calls=[[100.0 + i, "Bash", "npm test", False, "1 failed", "h1"] for i in range(4)])
        c = self.fail(errors=log)
        self.assertIn("Stuck", c["title"])
        self.assertIn("npm test", c["happened"])
        self.assertIn("fails twice, stop", c["fix"]["snippet"])

    def test_the_last_command_timed_out(self):
        log = errlog(calls=[[500.0, "Bash", "npm run e2e", True, "", "h2"]],
                     timeouts=[{"at": 500.0, "tool": "Bash", "target": "npm run e2e", "after_s": 120.0,
                                "background": False}])
        c = self.fail(errors=log)
        self.assertIn("timeout", c["title"].lower())
        self.assertIn('"BASH_DEFAULT_TIMEOUT_MS": "300000"', c["fix"]["snippet"])

    def test_a_timeout_moved_to_the_background_is_not_the_cause(self):
        log = errlog(calls=[[500.0, "Bash", "npm run e2e", True, "", "h2"]],
                     timeouts=[{"at": 500.0, "tool": "Bash", "target": "npm run e2e", "after_s": 120.0,
```

```python
"""What to change next time: fix-it cards for a run, and a one-line verdict.

Rule-based and explainable, like orchestra.runaway and orchestra.verify. Every card says what
happened (the evidence, with the agents it concerns), what it cost, and one concrete change with
where it goes. "possible" marks a judgment the record cannot settle; "sure" is a fact in it.
Cuelight never applies a fix: a card's snippet is text to copy.

Pure function of a Run (plus the price table and the usual numbers the builder attached to each
agent): no I/O, bounded output, every transcript-derived string scrubbed. Cuelight's own text is
ASCII only, so the command line can print it on any console.
"""

import hashlib
from typing import Any, Dict, List, Optional

from orchestra import constants as C
from orchestra import pressure
from orchestra.redact import scrub
from orchestra.runaway import detect_loop

MIN_SECONDS = 60.0        # a card that cost less than all three of these is noise
MIN_TOKENS = 10_000
MIN_COST = 0.05
MAX_CARDS = 12
API_RECENT_S = 120.0      # an API error this close to an agent's end is why it ended
CONTEXT_FULL = 0.95       # a context this full (of its assumed window) ran out
DONE_BADLY = (C.FAILED, C.ORPHANED, C.STALLED)

WHERE = {"settings": "settings.json", "agent": "the agent's definition", "claude_md": "CLAUDE.md",
         "instructions": "the agent's instructions", "habit": "a habit", "none": "nothing to change"}

def fmt_duration(seconds: Optional[float]) -> str:
    if seconds is None:
        return "-"
    total = int(round(max(0.0, seconds)))
    if total < 60:
        return "{}s".format(total)
    if total < 3600:
        return "{}m {:02d}s".format(total // 60, total % 60)
    return "{}h {:02d}m".format(total // 3600, (total % 3600) // 60)


def fmt_money(amount: Optional[float], currency: Optional[str] = None) -> str:
    if amount is None:
        return "-"
    symbol = "$" if not currency or currency == "USD" else currency + " "
    if 0 < amount < 0.01:
        return "<" + symbol + "0.01"
    return symbol + ("{:.0f}".format(amount) if amount >= 100 else "{:.2f}".format(amount))


def fmt_count(n: int) -> str:
    if n >= 1_000_000:
        return "{:.1f}M".format(n / 1_000_000)
    if n >= 1000:
        return "{:.1f}k".format(n / 1000)
    return str(n)


def _plural(n: int, one: str, many: str = "") -> str:
    return "{} {}".format(n, one if n == 1 else (many or one + "s"))


def _label(agent: Any) -> str:
    return scrub(agent.description)[:60] or agent.agent_id[:12]


def _fresh(tokens: Optional[Dict[str, int]]) -> int:
    return sum(v for k, v in (tokens or {}).items() if k != "cache_read")


# ------------------------------------------------------------------ cards

def _card(key: str, rule: str, tag: str, severity: str, title: str, happened: str, agents: List[str],
          where: str, text: str, snippet: str = "", seconds: float = 0.0, tokens: int = 0,
          cost: Optional[float] = None, label: str = "", title_many: str = "",
          always: bool = False) -> Dict[str, Any]:
    """One card before merging. `key` decides what merges; `title_many` ("{n} agents ...") names a
    merged card; `always` shows it whatever its size (its cost is a risk, not time or money)."""
    return {"key": key, "rule": rule, "tag": tag, "severity": severity, "title": title,
            "title_many": title_many or title, "happened": happened, "agents": list(agents),
            "impact": {"seconds": seconds, "tokens": tokens, "cost": cost, "label": label},
            "fix": {"where": where, "text": text, "snippet": snippet}, "always": always, "more": 0}


def _merge(cards: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    """Cards with the same key become one: agents joined, impacts added."""
    out: Dict[str, Dict[str, Any]] = {}
    for card in cards:
        have = out.get(card["key"])
        if have is None:
            out[card["key"]] = dict(card, agents=list(card["agents"]), impact=dict(card["impact"]),
                                    fix=dict(card["fix"]))
            continue
        have["agents"] += [a for a in card["agents"] if a not in have["agents"]]
        have["more"] += 1
        have["always"] = have["always"] or card["always"]
        imp, add = have["impact"], card["impact"]
        imp["seconds"] += add["seconds"]
        imp["tokens"] += add["tokens"]
        if add["cost"] is not None:
            imp["cost"] = (imp["cost"] or 0.0) + add["cost"]
    for card in out.values():
        if card["more"]:
            card["title"] = card["title_many"].format(n=card["more"] + 1)
            card["happened"] += " (and {} more like it)".format(card["more"])
    return list(out.values())


def _worth_showing(card: Dict[str, Any]) -> bool:
    if card["always"]:
        return True
    imp = card["impact"]
    return imp["seconds"] >= MIN_SECONDS or imp["tokens"] >= MIN_TOKENS or (imp["cost"] or 0.0) >= MIN_COST


def _order(card: Dict[str, Any]):
    imp = card["impact"]
    return (card["severity"] != "failure", -(imp["cost"] or 0.0), -imp["seconds"], -imp["tokens"], card["key"])


def _finish(card: Dict[str, Any]) -> Dict[str, Any]:
    return {"id": hashlib.sha1(card["key"].encode("utf-8")).hexdigest()[:12], "rule": card["rule"],
            "tag": card["tag"], "severity": card["severity"], "title": card["title"],
            "happened": card["happened"], "agents": card["agents"], "impact": card["impact"], "fix": card["fix"]}


def _select(cards: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
    """Merged, filtered, ordered and capped: what the reader sees."""
    merged = _merge(cards)
    return [_finish(c) for c in sorted((c for c in merged if _worth_showing(c)), key=_order)[:MAX_CARDS]]

API_FIX = {
    "rate_limit": ("a usage limit", "Wait for the limit to reset, then resume the agent. Runs started just "
                                    "before a reset are the ones that hit it."),
    "authentication_failed": ("an expired login", "Run /login, then resume the agent."),
    "server_error": ("a server error", "The API failed on Anthropic's side. Run the agent again."),
    "overloaded": ("an overloaded API", "The API was overloaded. Run the agent again in a few minutes."),
}


def _failure(agent: Any, now: float) -> Dict[str, Any]:
    """The first cause the record shows for an agent that failed, was cut off or went quiet."""
    label = _label(agent)
    verb = {C.STALLED: "went quiet", C.ORPHANED: "was cut off"}.get(agent.status, "failed")
    end = agent.ended_at or agent.last_activity_at or now
    log = agent.errors
    spent = {"seconds": agent.duration_s or max(0.0, end - (agent.started_at or end)),
             "tokens": _fresh(agent.tokens), "cost": agent.cost, "label": "spent before it stopped"}

    def card(cause: str, title: str, title_many: str, happened: str, where: str, text: str,
             snippet: str = "", merge: bool = False) -> Dict[str, Any]:
        key = "failure:{}:{}".format(cause, "" if merge else agent.agent_id)
        return _card(key, "failure", "sure", "failure", title, happened, [agent.agent_id], where, text,
                     snippet, title_many=title_many, always=True, **spent)

    last_api = log.api[-1] if log is not None and log.api else None
    if last_api is not None and last_api.get("at") is not None and last_api["at"] >= end - API_RECENT_S:
        kind = last_api.get("kind") or "error"
        what, fix = API_FIX.get(kind, ("an API error", "Run the agent again. If it happens again, read the "
                                                       "error above."))
        message = scrub(last_api.get("text") or "")[:160]
        return card("api-" + kind, "Stopped by " + what, "{n} agents stopped by " + what,
                    "{} {} right after the API returned {}{}.".format(label, verb, what,
                                                                     ": " + message if message else ""),
                    "habit", fix, merge=True)
    stuck = log.stuck() if log is not None else None
    if stuck is not None:
        target = scrub(stuck["target"])[:120]
        error = scrub(stuck.get("error") or "")[:160]
        return card("stuck", "Stuck on a failing call", "{n} agents stuck on a failing call",
                    "{} ran {} {} {} times in a row and it failed every time{}.".format(
                        label, stuck["tool"], target, stuck["failed_in_a_row"], ": " + error if error else ""),
                    "instructions", "Add a line to this agent's instructions (its brief, or its definition in "
                                    ".claude/agents/) so it stops instead of retrying:",
                    "If {} on {} fails twice, stop and report the error and what you tried.".format(stuck["tool"],
                                                                                                  target))
    done = log.done_calls() if log is not None else []
    if done:
        timeout = next((t for t in reversed(log.timeouts)
                        if not t.get("background") and t.get("at") == done[-1][0]), None)
        if timeout is not None:
            limit_ms = int(max(timeout["after_s"] * 2, 300) * 1000)
            return card("timeout", "Its last command ran past its timeout",
                        "{n} agents' last command ran past its timeout",
                        "{}'s last command, {}, was stopped after {}.".format(
                            label, scrub(timeout["target"])[:120], fmt_duration(timeout["after_s"])),
                        "settings", "Ask for long commands to run in the background, or raise the Bash timeout "
                                    "for this project in .claude/settings.json:",
                        '{"env": {"BASH_DEFAULT_TIMEOUT_MS": "%d"}}' % limit_ms)
    peak = pressure.peak_of(agent.waste)
    compactions = len(agent.waste.compactions) if agent.waste is not None else 0
    if compactions or (peak is not None and peak["fill"] >= CONTEXT_FULL):
        fill = "{:.0f}%".format(100 * peak["fill"]) if peak is not None else "most"
        return card("context", "Ran out of context", "{n} agents ran out of context",
                    "{}'s context reached {} of its window{}.".format(
                        label, fill, " and was compacted " + _plural(compactions, "time") if compactions else ""),
                    "instructions", "Split this task across two agents, each with a smaller part, so neither "
                                    "fills its context.")
    loop = detect_loop(agent.tool_calls, C.LOOP_REPEATS, C.LOOP_CYCLE_CALLS)
    if loop is not None:
        first = loop.calls[0]
        what = ("{} {} {} times in a row".format(first[0], scrub(first[1])[:120], loop.count)
                if loop.kind == "repeat" else "the same {} calls over and over".format(len(loop.calls)))
        return card("loop", "Went in circles", "{n} agents went in circles",
                    "{} repeated {} before it {}.".format(label, what, verb), "instructions",
                    "Give it a stop condition in its instructions:",
                    "If the same step fails or repeats three times, stop and report what you tried.")
    open_call = bool(agent.tool_calls) and agent.tool_calls[-1].ok is None
    if agent.status == C.ORPHANED or open_call:
        return card("closed", "Stopped when the session ended", "{n} agents stopped when the session ended",
                    "{} {} while {}.".format(label, verb, "a tool call was still open" if open_call
                                             else "its session ended"),
                    "none", "Nothing to fix in the setup. Resume it if its work is still needed.", merge=True)
    recent = "; ".join("{} {}".format(c.name, scrub(c.target)[:60]).strip() for c in agent.tool_calls[-3:])
    said = next((line.strip() for line in scrub(agent.result or "").splitlines() if line.strip()), "")[:160]
    return card("unknown", "No cause in the record", "{n} agents stopped with no cause in the record",
                "{} {}. Its last calls: {}.{}".format(label, verb, recent or "none",
                                                     " Its last message: " + said if said else ""),
                "none", "Open the agent to read what it did last.")


def _failure_cards(run: Any, now: float, table: Any) -> List[Dict[str, Any]]:
    return [_failure(a, now) for a in run.agents if a.status in DONE_BADLY]


# Every rule: fn(run, now, table) -> cards. Later rules are added to this tuple.
RULES = (_failure_cards,)


# ------------------------------------------------------------------ verdict

def _verdict_parts(run: Any, cards: List[Dict[str, Any]]) -> List[str]:
    totals = run.totals()
    n, failed = totals["agents"], totals.get(C.FAILED, 0) + totals.get(C.ORPHANED, 0)
    parts = [(_plural(n, "agent") + (", {} failed".format(failed) if failed else "")) if n else "No agents"]
    out = (run.insights or {}).get("outcomes") or {}
    made = [_plural(out[k], word) for k, word in (("commits", "commit"), ("prs", "PR")) if out.get(k)]
    if made:
        parts.append(", ".join(made))
    checks = out.get("checks") or {}
    passed, bad = checks.get("passed", 0), checks.get("failed", 0)
    if bad:
        parts.append("{} of {} test runs failed".format(bad, passed + bad))
    elif passed:
        parts.append("tests passed")
    cost = run.cost or {}
    if cost.get("enabled") and cost.get("total") is not None:
        parts.append(fmt_money(cost["total"], cost.get("currency")) +
                     (" (list price)" if cost.get("source") == "list" else ""))
    parts.append(_plural(len(cards), "thing") + " to fix" if cards else "nothing to fix")
    return parts


def compute(run: Any, now: float, table: Any = None) -> Dict[str, Any]:
    """The Advice payload for a run. `table` is the PriceTable in use, or None."""
    raw: List[Dict[str, Any]] = []
    for rule_fn in RULES:
        raw.extend(rule_fn(run, now, table))
    cards = _select(raw)
    cost = run.cost or {}
    enabled = bool(cost.get("enabled"))
    return {"parts": _verdict_parts(run, cards), "cards": cards, "slow": [],
            "failures": sum(1 for c in cards if c["severity"] == "failure"),
            "priced_by": cost.get("source") if enabled else None,
            "as_of": cost.get("as_of") if enabled and cost.get("source") == "list" else None}
```

```python
def _seen_check(run: Any) -> str:
    """The latest test, build or lint command any agent ran: the one to ask for."""
    best = None
    for agent in run.agents:
        for call in agent.tool_calls:
            if call.verify == verify.CHECK and call.timestamp is not None and \
                    (best is None or call.timestamp > best.timestamp):
                best = call
    return scrub(best.target)[:120] if best is not None else ""


def _verify_cards(run: Any, now: float, table: Any) -> List[Dict[str, Any]]:
    out: List[Dict[str, Any]] = []
    seen: Optional[str] = None
    for agent in run.agents:
        v = agent.verification
        if not v or not v.get("final") or v.get("state") not in (verify.UNCHECKED, verify.FAILING):
            continue
        if seen is None:
            seen = _seen_check(run)
```

```python
# Commands worth pre-approving: tests, builds, linters. Anything else is reported, never suggested:
# allowing it for good would also allow commands the user has not seen.
SAFE_PREFIXES = tuple(tuple(p.split()) for p in (
    "npm test", "npm run test", "npm run lint", "npm run build", "npm run typecheck",
    "pnpm test", "pnpm run test", "pnpm run lint", "pnpm run build", "yarn test", "yarn lint", "yarn build",
    "pytest", "python -m pytest", "python -m unittest", "python3 -m pytest", "python3 -m unittest",

UNSAFE_MARKS = (";", "&", "|", ">", "<", "`", "$(", "\n")

def safe_prefix(command: str) -> Optional[str]:
    """The allow-listed prefix a command starts with, or None. A compound command, a redirect or a
    substitution never gets one: a prefix rule for it would approve more than it says."""
    text = (command or "").strip()
    if not text or any(mark in text for mark in UNSAFE_MARKS):
        return None
    words = tuple(text.split())
    for prefix in sorted(SAFE_PREFIXES, key=len, reverse=True):
        if words[:len(prefix)] == prefix:
            return " ".join(prefix)
    return None


def _waited_on(run: Any, agent_id: str, start: float) -> Optional[List[Any]]:
    """The call a prompt was about: the latest one the agent made at or before it."""
    if agent_id:
        owner = run.agent(agent_id)
        log = owner.errors if owner is not None else None
    else:
        log = run.main_errors
    best = None
    for call in (log.calls if log is not None else []):
        if call[0] is not None and call[0] <= start + 1.0 and (best is None or call[0] >= best[0]):
            best = call
    return best


def _preapprove_cards(run: Any, now: float, table: Any) -> List[Dict[str, Any]]:
    groups: Dict[str, Dict[str, Any]] = {}
    for wait in run.waits:
        if wait.kind != livestate.PERMISSION or wait.state == livestate.UNANSWERED:
            continue
        call = _waited_on(run, wait.agent_id, wait.start)
        tool, target = (call[1], call[2] or "") if call else ("a tool", "")
        shown, allow = "", None
        if tool == "Bash":

        g = groups.setdefault(key, {"tool": tool, "shown": shown, "allow": allow, "seconds": 0.0, "prompts": 0,
                                    "agents": []})
```

```python
READ_ONLY_TOOLS = frozenset(("Read", "Grep", "Glob", "LS", "WebFetch", "WebSearch", "NotebookRead", "TodoWrite"))
MIN_READ_CALLS = 3
CHEAPER_SHARE = 0.30      # suggest a cheaper model only when it saves at least this share
PARALLEL_GAP_S = 120.0    # launched this soon after the previous one ended: it was waiting on it


def _family(model: str) -> str:
    name = (model or "").lower()
    return next((f for f in listprices.CHEAPER if f in name), "")


def _cheaper_cards(run: Any, now: float, table: Any) -> List[Dict[str, Any]]:
    if table is None:
        return []
    out = []
    for agent in run.agents:
        if agent.status != C.COMPLETED or len(agent.tool_calls) < MIN_READ_CALLS:
            continue
        if any(call.name not in READ_ONLY_TOOLS for call in agent.tool_calls):
            continue
        family = _family(agent.model)
        if not family:
            continue
        alias, target = listprices.CHEAPER[family]
        prices = table.price_for(target)
        cost, unpriced = table.cost(agent.tokens_by_model)
        if prices is None or unpriced or cost <= 0:
            continue
        alt = sum(count * prices.get(kind, 0.0) for tokens in agent.tokens_by_model.values()
                  for kind, count in tokens.items()) / 1_000_000
        saving = cost - alt
        if saving < MIN_COST or saving < cost * CHEAPER_SHARE:
            continue
        out.append(_card("cheaper_model:{}:{}".format(scrub(agent.agent_type) or "?", family), "cheaper_model",
                         "possible", "fix", "A cheaper model would likely have done",
                         "{} only read and searched ({} calls, no edits or commands) on {}. It cost {}; the same "
                         "tokens on {} would cost about {}.".format(
                             _label(agent), len(agent.tool_calls), _short(agent.model),
                             fmt_money(cost, table.currency), alias, fmt_money(alt, table.currency)),
                         [agent.agent_id], "agent",
                         "If this kind of agent has a definition in .claude/agents/, set its model there; otherwise "
                         "ask Claude to launch it with model {}. Check that its answers still hold up.".format(alias),
                         "model: {}".format(alias), cost=saving, label="could save",
                         title_many="{n} read-only agents ran on an expensive model"))
    return out


def _linked(a: Any, b: Any, edges: set) -> bool:
    """Whether one agent could have needed the other: an edge, or a file one wrote that the other used."""
    if frozenset((a.agent_id, b.agent_id)) in edges:
        return True
    wa, ra = {normalize_path(p) for p in a.files_written}, {normalize_path(p) for p in a.files_read}
    wb, rb = {normalize_path(p) for p in b.files_written}, {normalize_path(p) for p in b.files_read}
    return bool(wa & rb or wb & ra or wa & wb)


def _chain_card(chain: List[Any]) -> List[Dict[str, Any]]:
    if len(chain) < 2:
        return []
    durations = [a.duration_s or 0.0 for a in chain]
    names = ", ".join(_label(a) for a in chain[:4]) + (" and {} more".format(len(chain) - 4) if len(chain) > 4 else "")
    return [_card("parallel:" + ",".join(a.agent_id for a in chain), "parallel", "possible", "fix",
                  "These agents could have run together",
                  "{} agents ran one after another ({}) and none used another's work: no handoff, no shared "
                  "file.".format(len(chain), names),
                  [a.agent_id for a in chain], "habit",
                  "When agents don't depend on each other, ask Claude to launch them in one message so they run "
                  "at the same time.", seconds=sum(durations) - max(durations), label="could save")]


def _parallel_cards(run: Any, now: float, table: Any) -> List[Dict[str, Any]]:
    turns = run.main_turns.turns if run.main_turns is not None else []
    prompts = sorted(t["at"] for t in turns if t.get("at") is not None)
    edges = {frozenset((e.src, e.dst)) for e in run.edges}
    batch = {aid: i for i, b in enumerate(run.batches) for aid in b.agent_ids}
    groups: Dict[str, List[Any]] = {}
    for a in run.agents:
        if a.status == C.COMPLETED and a.started_at is not None and a.ended_at is not None:
            groups.setdefault(a.parent_agent_id or "", []).append(a)
    out: List[Dict[str, Any]] = []
    for agents in groups.values():
        agents.sort(key=lambda a: a.started_at)
        chain: List[Any] = []
        for agent in agents:
            prev = chain[-1] if chain else None
            joins = (prev is not None and agent.started_at >= prev.ended_at
                     and agent.started_at - prev.ended_at <= PARALLEL_GAP_S
                     and not any(prev.ended_at < t <= agent.started_at for t in prompts)
                     and (batch.get(agent.agent_id) is None or batch.get(agent.agent_id) != batch.get(prev.agent_id))
                     and not any(_linked(m, agent, edges) for m in chain))
            if not joins:
                out += _chain_card(chain)
                chain = []
            chain.append(agent)
        out += _chain_card(chain)
```

```python
def _impact_text(impact: Dict[str, Any], currency: Optional[str]) -> str:
    parts = []
    if impact.get("cost"):
        parts.append(fmt_money(impact["cost"], currency))
    if (impact.get("seconds") or 0) >= 1:
        parts.append(fmt_duration(impact["seconds"]))
    if impact.get("tokens"):
        parts.append(fmt_count(impact["tokens"]) + " tokens")
    return (impact["label"] + ": " if parts and impact.get("label") else "") + ", ".join(parts)


def fact_sheet(run: Any) -> List[str]:
    """A short, scrubbed account of the run, so Claude can answer follow-up questions from facts."""
    totals = run.totals()
    currency = (run.cost or {}).get("currency")
    lines = ["wall time: {}".format(fmt_duration(totals["wall_time_s"])), "agents: {}".format(totals["agents"])]
    for agent in run.agents[:40]:
        lines.append("- {} | {} | {} | {} | {} | {} | {} tokens{} | {} calls | {} files written | checks: {}".format(
            agent.agent_id[:10], _label(agent), scrub(agent.agent_type) or "?", _short(agent.model), agent.status,
            fmt_duration(agent.duration_s), fmt_count(_fresh(agent.tokens)),
            " | " + fmt_money(agent.cost, currency) if agent.cost is not None else "",
            len(agent.tool_calls), len(agent.files_written), (agent.verification or {}).get("state") or "-"))
    if len(run.agents) > 40:
        lines.append("- ... and {} more agents".format(len(run.agents) - 40))
    tools: Dict[str, int] = {}
    for agent in run.agents:
        for call in agent.tool_calls:
            tools[call.name] = tools.get(call.name, 0) + 1
    top = sorted(tools.items(), key=lambda kv: (-kv[1], kv[0]))[:5]
    if top:
        lines.append("top tools: " + ", ".join("{} x{}".format(name, count) for name, count in top))
    out = (run.insights or {}).get("outcomes") or {}
    lines.append("write conflicts: {} | links between agents: {} | commits: {} | pull requests: {}".format(
        len(run.write_conflicts), len(run.edges), out.get("commits", 0), out.get("prs", 0)))
    return lines[:60]


def render_text(run: Any) -> str:
    """What /cuelight:open advise prints: the verdict, every card with its fix, then the fact sheet."""
    a = run.advice or {"parts": [], "cards": [], "as_of": None}
    currency = (run.cost or {}).get("currency")
    lines = ["Cuelight advice", "", " | ".join(a["parts"])]
    if a.get("as_of"):
        lines.append("Dollar figures use Anthropic list prices as of {}.".format(a["as_of"]))
    lines.append("")
    if not a["cards"]:
        lines += ["Nothing to fix in this run.", ""]
    for i, card in enumerate(a["cards"], 1):
        lines.append("{}. {} [{}]".format(i, card["title"], card["tag"]))
        lines.append("   What happened: " + card["happened"])
        cost = _impact_text(card["impact"], currency)
        if cost:
            lines.append("   What it cost: " + cost)
        fix = card["fix"]
        lines.append("   Fix ({}): {}".format(WHERE.get(fix["where"], fix["where"]), fix["text"]))
        lines += ["      " + line for line in fix["snippet"].splitlines()]
        lines.append("")
    lines.append("Run facts, for follow-up questions:")
    lines += fact_sheet(run)
    return "\n".join(lines)
```

```js
// what happened, what it cost and the change to make, which you copy. Cuelight never applies one.
//
// Loaded after app.js and tabs.js, whose globals (state, $, esc, fmtDuration, fmtCount, fmtMoney,
// plural, priceNote, openDrawer, noteFocus, restoreFocus) it uses; app.js's render() calls
// renderAdvice and renderAdviceBadge. A static report inlines it after tabs.js.

const ADVICE_HIDDEN_KEY = "cuelight-advice-hidden";
const ADVICE_WHERE = {
  settings: "In settings.json:", agent: "In the agent's definition:", claude_md: "In CLAUDE.md:",
  instructions: "In the agent's instructions:", habit: "A habit:", none: "",
};

// Hidden cards are a per-browser convenience: a private window simply forgets them.
function adviceHidden() {
  try {
    const raw = typeof localStorage !== "undefined" ? localStorage.getItem(ADVICE_HIDDEN_KEY) : null;
    const ids = raw ? JSON.parse(raw) : [];
    return Array.isArray(ids) ? ids : [];
  } catch (err) { return []; }
}

function setAdviceHidden(ids) {
  try { if (typeof localStorage !== "undefined") localStorage.setItem(ADVICE_HIDDEN_KEY, JSON.stringify(ids.slice(-200))); }
  catch (err) { /* hiding lasts until the page reloads */ }
}

function adviceImpact(impact, currency) {
  const parts = [];
  if (impact.cost) parts.push(fmtMoney(impact.cost, currency));
  if (impact.seconds >= 1) parts.push(fmtDuration(impact.seconds));
  if (impact.tokens) parts.push(fmtCount(impact.tokens) + " tokens");
  if (!parts.length) return "";
  return '<span class="advice-impact">' + esc((impact.label ? impact.label + ": " : "") + parts.join(", ")) + "</span>";
}

function adviceAgents(card, run) {
  const names = (card.agents || []).map((id) => {
    if (!id) return '<span class="advice-agent">Main session</span>';
    const agent = (run.agents || []).find((a) => a.agent_id === id);
    return '<button type="button" class="advice-agent" data-agent="' + esc(id) + '">' +
      esc((agent && agent.description) || id) + "</button>";
  });
  return names.length ? '<p class="advice-agents">' + names.join("") + "</p>" : "";
}

function adviceCardHtml(card, run, currency, isHidden) {
  const fix = card.fix || {};
  const where = ADVICE_WHERE[fix.where] || "";
  return '<article class="advice-card" data-severity="' + esc(card.severity) + '">' +
    '<header><span class="advice-tag" data-tag="' + esc(card.tag) + '">' + esc(card.tag) + "</span>" +

    '<button type="button" class="advice-hide" data-hide="' + esc(card.id) + '">' + (isHidden ? "Unhide" : "Hide") +
    "</button></header>" +
    '<p class="advice-happened">' + esc(card.happened) + "</p>" + adviceAgents(card, run) +
    '<div class="advice-fix"><p>' + (where ? '<span class="advice-where">' + esc(where) + "</span> " : "") +
    esc(fix.text || "") + "</p>" +
    (fix.snippet ? '<pre class="advice-snippet">' + esc(fix.snippet) + '</pre><button type="button" class="advice-copy" data-copy="' +
      esc(card.id) + '">Copy</button>' : "") + "</div></article>";
}

function adviceHtml(run, hidden, showHidden) {
  if (run.replay_at !== undefined) return '<p class="advice-note">Advice describes the whole session. Leave replay to see it.</p>';
  const a = run.advice;
  if (!a) return '<p class="advice-note">This run has no advice: it was recorded by an older Cuelight.</p>';
  const currency = run.cost && run.cost.currency;
  const note = priceNote(run.cost);
  const cards = a.cards || [];
  const hiddenHere = cards.filter((c) => hidden.indexOf(c.id) >= 0).length;
  const shown = showHidden ? cards : cards.filter((c) => hidden.indexOf(c.id) < 0);
  let html = '<section class="advice-verdict"><p>' + esc((a.parts || []).join(" \u00B7 ")) + "</p>" +
    '<button type="button" class="advice-copy" data-copy="verdict">Copy</button>' +
    (note ? '<p class="advice-price">Dollar figures: ' + esc(note) + ".</p>" : "") + "</section>";
  html += shown.length
    ? shown.map((c) => adviceCardHtml(c, run, currency, hidden.indexOf(c.id) >= 0)).join("")
    : '<p class="advice-note">' + (cards.length ? "Every card is hidden." : "Nothing to fix in this run.") + "</p>";
  if (hiddenHere) {
    html += '<button type="button" class="advice-show-hidden" data-show-hidden="' + (showHidden ? "0" : "1") + '">' +
      (showHidden ? "Leave the hidden cards out" : "Show " + plural(hiddenHere, "hidden card", "hidden cards")) + "</button>";
  }
  return html;
}

async function copyAdviceText(text, btn) {
  let ok = false;
  try {
    if (typeof navigator !== "undefined" && navigator.clipboard && navigator.clipboard.writeText) {
      await navigator.clipboard.writeText(text);
      ok = true;
    }
  } catch (err) { /* ok stays false; the button says so below */ }
  const original = btn.textContent;
  btn.textContent = ok ? "Copied!" : "Copy failed";
  setTimeout(() => { btn.textContent = original; }, 1500);
}
```

```python
"""What each kind of agent usually takes: time, tokens, cost and how often it fails, learned from
your own past sessions.

Claude Code keeps every session's transcripts (about 30 days by default). This reads them, read-only,
with the same RunBuilder the dashboard uses, and keeps NUMBERS ONLY per finished agent: no prompts,
briefs, results, paths or commands; the project is a short hash of its folder name. The rows are
cached in Cuelight's state directory (never ~/.claude), one entry per session, valid while the
session's files are unchanged, so after the first scan only new or changed sessions are read. The
cache is a convenience: deleting it costs one rescan, and a corrupt or old one is ignored.

"Usual" is per agent type and model family (most agents are general-purpose, so the model is what
tells them apart): this project's runs when there are at least MIN_RUNS, else every project's,
else "not enough past runs yet". Typical is the median, slow case the 90th percentile.
"""

import glob
import hashlib
import json
import math
import os
import statistics
import tempfile
import threading
import time
from typing import Any, Dict, List, Optional, Tuple

from orchestra import constants as C
from orchestra import locate
from orchestra.advice import fmt_count, fmt_duration, fmt_money
from orchestra.build import RunBuilder
from orchestra.runaway import detect_loop
from orchestra.spend import _short
from orchestra.statedir import state_dir

VERSION = 1
CACHE_NAME = "baselines-v1.json"
MIN_RUNS = 5
RESCAN_S = 600.0          # a dashboard left open picks up new sessions this often
MAX_GROUPS = 40
FINISHED = (C.COMPLETED, C.FAILED, C.ORPHANED)


def project_key(project_dir: str) -> str:
    """A project by a short hash of its folder name: matching needs no path kept."""
    name = os.path.basename((project_dir or "").rstrip("/\\"))
    return hashlib.sha1(name.encode("utf-8")).hexdigest()[:12] if name else ""


def _fresh(tokens_by_model: Dict[str, Dict[str, int]]) -> int:
    return sum(v for tokens in tokens_by_model.values() for k, v in tokens.items() if k != "cache_read")


def _fingerprint(paths: locate.SessionPaths) -> List[List[Any]]:
    files = [paths.session_jsonl] + sorted(glob.glob(os.path.join(paths.subagents_dir, "agent-*")))
    out = []
    for path in files:
        try:
            st = os.stat(path)
        except OSError:
            continue
        out.append([os.path.basename(path), st.st_size, st.st_mtime_ns])
    return out


def rows_for(paths: locate.SessionPaths) -> List[Dict[str, Any]]:
    """Numbers for every finished agent of one session."""
    run = RunBuilder(paths).refresh()
    project = project_key(paths.project_dir)
    out = []
    for a in run.agents:
        if a.status not in FINISHED or a.duration_s is None:
            continue
        out.append({"session": paths.session_id, "project": project, "type": a.agent_type or "?",
                    "model": _short(a.model), "status": a.status, "start": a.started_at,
                    "seconds": a.duration_s, "tokens": {m: dict(t) for m, t in a.tokens_by_model.items()},
                    "calls": len(a.tool_calls), "edited": a.verification is not None,
                    "checked": (a.verification or {}).get("state", ""),
                    "loop": detect_loop(a.tool_calls, C.LOOP_REPEATS, C.LOOP_CYCLE_CALLS) is not None,
                    "waited": a.waited_s})
    return out


def _p90(values: List[float]) -> float:
    ordered = sorted(values)
    return ordered[max(0, math.ceil(0.9 * len(ordered)) - 1)]


def _summarize(rows: List[Dict[str, Any]], scope: str, table: Any) -> Dict[str, Any]:
    seconds = [r["seconds"] for r in rows]
    fresh = [_fresh(r["tokens"]) for r in rows]
    costs = [table.cost(r["tokens"])[0] for r in rows] if table is not None else None
    edited = [r for r in rows if r["edited"]]
    return {"type": rows[0]["type"], "model": rows[0]["model"], "scope": scope, "n": len(rows), "enough": True,
            "typical_s": statistics.median(seconds), "slow_s": _p90(seconds),
            "typical_tokens": int(statistics.median(fresh)), "slow_tokens": int(_p90(fresh)),
            "typical_cost": statistics.median(costs) if costs else None,
            "slow_cost": _p90(costs) if costs else None,
            "fail_rate": sum(1 for r in rows if r["status"] != C.COMPLETED) / len(rows),
            "unchecked_rate": (sum(1 for r in edited if r["checked"] != "checked") / len(edited)) if edited else None}


class Baselines:
    """The past-runs store. One per server; every builder asks it, so reads are under a lock and a
    scan swaps in its result at the end."""

    def __init__(self, root: Optional[str] = None, cache_path: Optional[str] = None) -> None:
        self.root = root
        self._cache_path = cache_path
        self._lock = threading.Lock()
        self._sessions: Dict[str, Dict[str, Any]] = {}
        self._rows: List[Dict[str, Any]] = []
        self._thread: Optional[threading.Thread] = None
        self._scanned_at = 0.0
        self.state = "idle"       # idle | scanning | ready | error
        self.error = ""

    # -- scanning ------------------------------------------------------------

    def cache_path(self) -> str:
        return self._cache_path or os.path.join(state_dir(), CACHE_NAME)

    def _load(self) -> Dict[str, Dict[str, Any]]:
        try:
            with open(self.cache_path(), encoding="utf-8") as fh:
                data = json.load(fh)
        except (OSError, ValueError):
            return {}
        if not isinstance(data, dict) or data.get("version") != VERSION or not isinstance(data.get("sessions"), dict):
            return {}
        return data["sessions"]

    def _save(self, sessions: Dict[str, Dict[str, Any]]) -> None:
        path = self.cache_path()
        try:
            fd, tmp = tempfile.mkstemp(prefix=".baselines-", dir=os.path.dirname(path) or ".")
        except OSError:
            return
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as fh:
                json.dump({"version": VERSION, "sessions": sessions}, fh)
            os.chmod(tmp, 0o600)
            os.replace(tmp, path)
        except OSError:
            try:
                os.remove(tmp)
            except OSError:
                pass

    def _install(self, sessions: Dict[str, Dict[str, Any]]) -> None:
        rows = [r for entry in sessions.values() for r in entry.get("rows", [])]
        with self._lock:
            self._sessions, self._rows = sessions, rows
            self.state, self.error = "ready", ""
            self._scanned_at = time.time()

    def scan(self) -> None:
        """Read every session whose files changed since the cache was written. Never raises."""
        with self._lock:
            if self.state != "ready":
                self.state = "scanning"
        cached = self._load()
        fresh: Dict[str, Dict[str, Any]] = {}
        projects = os.path.join(self.root if self.root is not None else locate.claude_root(), "projects")
        try:
            names = sorted(os.listdir(projects)) if os.path.isdir(projects) else []
            for name in names:
                project_dir = os.path.join(projects, name)
                for info in locate.list_sessions(project_dir):
                    paths = locate._paths_for(os.path.join(project_dir, info.session_id + ".jsonl"), info.session_id)
                    fp = _fingerprint(paths)
                    old = cached.get(info.session_id)
                    if isinstance(old, dict) and old.get("fp") == fp:
                        fresh[info.session_id] = old
                        continue
                    try:
                        rows = rows_for(paths)
                    except Exception:      # noqa: BLE001 - one unreadable session must not stop the scan
                        rows = []
                    fresh[info.session_id] = {"fp": fp, "rows": rows}
        except OSError as exc:
            with self._lock:
                self.state, self.error = "error", "past runs unreadable: {}".format(exc)
            return
        self._install(fresh)
        self._save(fresh)

    def start(self) -> None:
        """Scan in a background thread; a call while one is running does nothing."""
        with self._lock:
            if self._thread is not None and self._thread.is_alive():
                return
            self._thread = threading.Thread(target=self.scan, name="cuelight-baselines", daemon=True)
            self._thread.start()

    def start_if_stale(self) -> None:
        if self.state != "scanning" and time.time() - self._scanned_at >= RESCAN_S:
            self.start()

    # -- reading -------------------------------------------------------------

    def rows(self) -> List[Dict[str, Any]]:
        with self._lock:
            return list(self._rows)

    def _group(self, rows: List[Dict[str, Any]], project: str, table: Any) -> Dict[str, Any]:
        mine = [r for r in rows if r["project"] == project] if project else []
        if len(mine) >= MIN_RUNS:
            return _summarize(mine, "project", table)
        if len(rows) >= MIN_RUNS:
            return _summarize(rows, "all", table)
        first = rows[0] if rows else {}
        return {"type": first.get("type", ""), "model": first.get("model", ""), "scope": "none", "n": len(rows),
                "enough": False}

    def usual_for(self, agent_type: str, model: str, project_dir: str, exclude_session: str,
                  table: Any = None) -> Optional[Dict[str, Any]]:
        """What this kind of agent usually takes, or None before the first scan has finished."""
        if self.state != "ready":
            return None
        kind, family = agent_type or "?", _short(model)
        same = [r for r in self.rows() if r["type"] == kind and r["model"] == family and r["session"] != exclude_session]
        out = self._group(same, project_key(project_dir), table)
        out.update(type=kind, model=family)
        return out

    def summary(self, project_dir: str, exclude_session: str, table: Any = None) -> Dict[str, Any]:
        """Every (type, model) seen, for the Advice tab's table and /cuelight:open estimate."""
        rows = [r for r in self.rows() if r["session"] != exclude_session]
        groups: Dict[Tuple[str, str], List[Dict[str, Any]]] = {}
        for r in rows:
            groups.setdefault((r["type"], r["model"]), []).append(r)
        project = project_key(project_dir)
        out = []
        for (kind, family), group in groups.items():
            entry = self._group(group, project, table)
            entry.update(type=kind, model=family)
            out.append(entry)
        out.sort(key=lambda g: (not g["enough"], -g["n"], g["type"], g["model"]))
        return {"state": self.state, "error": self.error, "sessions": len({r["session"] for r in rows}),
                "agents": len(rows), "groups": out[:MAX_GROUPS],
                "source": table.source if table is not None else None,
                "as_of": table.as_of if table is not None else "",
                "currency": table.currency if table is not None else None}


def render_estimate(data: Dict[str, Any]) -> str:
    """What /cuelight:open estimate prints."""
    lines = ["Cuelight: what each kind of agent usually takes, from {} past sessions ({} finished agents).".format(
        data["sessions"], data["agents"]), ""]
    if data.get("error"):
        lines += [data["error"], ""]
    if data.get("source") == "list":
        lines.append("Costs use Anthropic list prices as of {}.".format(data["as_of"]))
    elif data.get("source") == "yours":
        lines.append("Costs use your price file.")
    else:
        lines.append("No prices: costs are left out.")
    groups = data.get("groups") or []
    if not groups:
        lines += ["", "No past runs yet. Cuelight learns what each kind of agent usually takes from your finished "
                      "sessions."]
        return "\n".join(lines)
    enough = [g for g in groups if g["enough"]]
    if enough:
        header = ("type", "model", "runs", "typical", "slow case", "tokens", "cost", "fails", "from")
        table = [header] + [(g["type"][:28], g["model"], str(g["n"]), fmt_duration(g["typical_s"]),
                             fmt_duration(g["slow_s"]), fmt_count(g["typical_tokens"]),
                             fmt_money(g["typical_cost"], data.get("currency")),
                             "{:.0f}%".format(100 * g["fail_rate"]),
                             "this project" if g["scope"] == "project" else "all projects") for g in enough]
        widths = [max(len(r[i]) for r in table) for i in range(len(header))]
        lines.append("")
        lines += ["  ".join(c.ljust(w) for c, w in zip(r, widths)).rstrip() for r in table]
    few = [g for g in groups if not g["enough"]]
    if few:
        lines += ["", "Not enough past runs yet (fewer than {}): ".format(MIN_RUNS) +
                  ", ".join("{} on {} ({})".format(g["type"], g["model"], g["n"]) for g in few[:12])]
    lines += ["", "Typical is the middle run; slow case is the 90th percentile. Use this to pick an agent type and "
                  "model before you launch one."]
    return "\n".join(lines)
```

```python
def _usual_line(u: Dict[str, Any]) -> str:
    return "{} on {} ({} runs, {})".format(u["type"], u["model"], u["n"],
                                          "this project" if u["scope"] == "project" else "your projects")


def _over_usual_cards(run: Any, now: float, table: Any) -> List[Dict[str, Any]]:
    out = []
    for agent in run.agents:
        u = agent.usual
        d = agent.duration_s
        if not u or not u.get("enough") or agent.status != C.COMPLETED or d is None:
            continue
        long = d > u["slow_s"] and d > OVER_RATIO * u["typical_s"]
        dear = (agent.cost is not None and u.get("typical_cost") and u.get("slow_cost") is not None
                and agent.cost > u["slow_cost"] and agent.cost > OVER_RATIO * u["typical_cost"])
        if not (long or dear):
            continue
        if long:
            happened = "{} took {}, {:.1f}x the usual {} for {}.".format(
                _label(agent), fmt_duration(d), d / u["typical_s"], fmt_duration(u["typical_s"]), _usual_line(u))
        else:
            happened = "{} cost {}, {:.1f}x the usual {} for {}.".format(
                _label(agent), fmt_money(agent.cost), agent.cost / u["typical_cost"], fmt_money(u["typical_cost"]),
                _usual_line(u))
        out.append(_card("over_usual:" + agent.agent_id, "over_usual", "sure", "info",
                         "Took far longer than usual" if long else "Cost far more than usual", happened,
                         [agent.agent_id], "none", "",
                         seconds=max(0.0, d - u["typical_s"]) if long else 0.0,
                         cost=(agent.cost - u["typical_cost"]) if dear else None, label="over usual"))
    return out


def _explain_over_usual(cards: List[Dict[str, Any]]) -> None:
    """An over-usual card points to the other cards on the same agent, which say why."""
    for card in cards:
        if card["rule"] != "over_usual":
            continue
        others = [c["title"] for c in cards if c is not card and set(c["agents"]) & set(card["agents"])]
        card["fix"]["text"] = ("Other cards here explain part of it: " + "; ".join(others) + "." if others else
                               "No other card explains it. Open the agent and look at its longest tool calls.")


def _slow(run: Any, now: float) -> List[Dict[str, Any]]:
    """Running agents far past what their kind usually takes: the Health box's SLOWER THAN USUAL."""
    out = []
    for agent in run.agents:
        u = agent.usual
        if not u or not u.get("enough") or agent.status not in (C.RUNNING, C.WAITING) or agent.started_at is None:
            continue
        elapsed = now - agent.started_at
        if elapsed > u["slow_s"] and elapsed > OVER_RATIO * u["typical_s"]:
            out.append({"agent_id": agent.agent_id, "elapsed_s": elapsed, "typical_s": u["typical_s"],
                        "slow_s": u["slow_s"], "n": u["n"], "kind": "{} on {}".format(u["type"], u["model"])})
    return out
```
